```python
import pandas as pd

def get_ranked_players(team_db_path, team_name, position):
    # Read team data from database
    team_data = pd.read_csv(team_db_path + '\\' + team_name + '.csv')
    
    # Filter team data for the specified position
    position_data = team_data[team_data['Position'] == position]
    
    # Rank players by their ratings
    ranked_players = position_data.reset_index(drop=True).sort_values(by='Rating', ascending=False)
    
    return ranked_players
```
